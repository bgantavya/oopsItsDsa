560. Subarray Sum Equals K

Given an array of integers nums and an integer k, return the total number of subarrays whose sum equals to k.

A subarray is a contiguous non-empty sequence of elements within an array.

---

## Intuition
so we have to find sub-arrays that can be made up of the nums array, 
---

## Approach
for that we will firstly found the sum of all elements iteratively stored that in a hashmap, and returned the value that came on 0

---

## Complexity

- **Time complexity:** $$O(n)$$
- **Space complexity:** $$O(n)$$

class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        int res = 0;
        int cur = 0;
        unordered_map<int,int>count;
        count[0] = 1;
        
        for(int i : nums){
            cur += i;
            res += count[cur-k];
            count[cur] =  count[cur] +1;
        }

        return res;
    }
};

        int res = 0;

        Dictionary<int, int> count = new Dictionary<int, int>();
        count[0] = 1;
        foreach (int i in nums){
            cur += i;
            if(count.ContainsKey(cur-k)) {
                res += count[cur-k];}

            if(count.ContainsKey(cur)) {
            count[cur] += 1;}
            else{
                count[cur] = 1;
            }
        }
        return res;
    }

class Solution {
    public int subarraySum(int[] nums, int k) {
        int res = 0;
        int cur = 0;
        Map<Integer, Integer> count = new HashMap<>();
        count.put(0,1);
        for(int i : nums){
            cur += i;
            res += count.getOrDefault(cur-k, 0);
            count.put(cur, count.getOrDefault(cur, 0) + 1);
        }
        return res;
    }
}
