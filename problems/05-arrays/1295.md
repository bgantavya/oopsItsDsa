```C []
int findNumbers(int* nums, int numsSize) {
    int count = 0;
    
    for (int i = 0; i < numsSize; i++) {
        int num = nums[i];
        int dg = 0;
        if (num == 0) dg = 1;
        
        while(num > 0){
        num /= 10;
        dg++; }

        if (dg % 2 == 0) count ++;
    }
    return count;
}
```

```JavaScript []
var findNumbers = function (nums) {
    let count = 0;
    for (let num of nums) {
        let i = 0;
        if (num == 0) i = 1;
        while (num > 0) {
            num = Math.floor(num / 10);
            i++;
        }
        if (i % 2 == 0) count++;
    }
    return count;
};
```
