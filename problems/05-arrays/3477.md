```C []
int numOfUnplacedFruits(int* fruits, int fruitsSize, int* baskets,
                        int basketsSize) {
    int count = 0;
    int unplaced = 0;
    bool used[basketsSize];
    for (int j = 0; j < basketsSize; j++) {
        used[j] = false;
    }

    for (int i = 0; i < fruitsSize; i++) {
        bool placed = false;
        for (int j = 0; j < basketsSize; j++) {

            if (!used[j] && baskets[j] >= fruits[i]) {
                placed = true;
                used[j] = true;
                break;
            }
        }
        if (!placed) {
            unplaced++;
        }
    }
    return unplaced;
}
```
