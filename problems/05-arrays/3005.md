```C []
int maxFrequencyElements(int* nums, int numsSize) {
    int arr[100];
    for (int i = 0; i < numsSize; i++)
        arr[nums[i] - 1]++;

    int max = 0;
    for (int i = 0; i < 100; i++)
        if (max < arr[i])
            max = arr[i];

    int count = 0;
    for (int i = 0; i < 100; i++)
        if (max == arr[i])
            count += arr[i];
    return count;
}
```
