import array
class Solution(object):
    def maximumPopulation(self, logs):
        """
        :type logs: List[List[int]]
        :rtype: int
        """
        
        years = array.array('i', [0]* 2051)
        for log in logs:
            years[log[0]] +=1
            years[log[1]] -=1
        
        maxYear = 1950
        currPop = 0
        maxPop = 0

        for i in range(1950, 2051):
            currPop += years[i]
            if(currPop > maxPop):
                maxPop = currPop
                maxYear = i
            
        return maxYear
