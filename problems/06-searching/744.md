```C []
char nextGreatestLetter(char* letters, int lettersSize, char target) {
    int a = 0;
    int b = lettersSize -1;
    while(a <= b){
        int mid = a + (b-a)/2;
        if(letters[mid] > target) b = mid-1;
        else a = mid+1;
    }
    return letters[a%lettersSize];
}
```

```Java []
        int mid;
        while(low <= high){
            mid = low + (high - low) /2;
```
