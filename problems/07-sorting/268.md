[ 268. Missing Number
Given an array nums containing n distinct numbers in the range [0, n], 
return the only number in the range that is missing from the array.

# Intuition
<!-- Describe your first thoughts on how to solve this problem. -->
since we have to find missing number where array nums of length n can have elements till n at indexes
so we can use cyclic sort to arrange then we can find the missing

# Approach
<!-- Describe your approach to solving the problem. -->
when we sort we put elements at their correct indexes which would be i + 1 = nums[i]
if there is any inconsistency we will return that element considering we did sort first

```C []
int missingNumber(int* nums, int numsSize) {
    int i = 0;
    while(i < numsSize){
        int correct = nums[i];
        if(correct < numsSize && nums[correct] != nums[i]){
            int temp = nums[correct];
            nums[correct] = nums[i];
            nums[i] = temp;
        }
        else i++;
    }
    for(i = 0; i < numsSize; i++){
        if (nums[i] != i) return i;
    }
    return numsSize;
}
```

```C++ []
class Solution {
public:
    int missingNumber(vector<int>& nums) {
        int i = 0;
        int n = nums.size();
        while(i < n){
            int correct = nums[i];
            if (correct < n && nums[i] != nums[correct]) swap(nums, i, correct);
            else i++;
        }
        for( i = 0; i < n; i++) if(i != nums[i]) return i;
        return n;
    }
    void swap(vector<int>& nums, int index, int correct){
        int temp = nums[correct];
        nums[correct] = nums[index];
        nums[index] = temp;
    }
};
```
